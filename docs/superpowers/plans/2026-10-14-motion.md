# Motion Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a pack-opening ceremony before the deal, a living shop shelf (glints, lift, sway, reroll spin), and a pointer-following glare with tilt shadow on the editor card — all display-only and off under reduced motion.

**Architecture:** A `usePackOpen` hook drives a phase machine (`sealed → shake → tear → burst → done`); `PackDeal` shows a `PackStage` (pack art split into top strip + body canvases) until `done`, then mounts the existing deal (moved into a `DealCards` child so `useDeal` starts only then). Shelf motion is CSS plus small prop/key changes in `Shop` and `TiltCard`. Editor glare is a `CardGlare` wrapper with a CSS overlay clipped around the photo window by a pure `glareClip(template)`.

**Tech Stack:** Vite 8, React 19.2, TypeScript ~6.0, Zustand 5, Vitest 4.1 (jsdom, vitest-canvas-mock), @testing-library/react 16.

**Spec:** `docs/superpowers/specs/2026-10-14-motion-design.md`

## Global Constraints

- No new dependencies; no dependency upgrades.
- Display only: `composeStamp`, export, print and download code paths unchanged.
- Everything new is off under `prefers-reduced-motion` (JS paths use `prefersReducedMotion()` from `src/ui/motion.ts`; CSS in the existing `@media (prefers-reduced-motion: reduce)` block).
- `OPEN_TIMING = { auto: 3000, shake: 350, tear: 400, burst: 600 }`; `TEAR_ROW = 9`; glint cycle 7 s with `--glint-delay` = index × 1.4 s; spin-in 350 ms staggered 80 ms; from-pack stagger 90 ms.
- Rare tint colours: Holographic `#7fd6ff`, Polychrome `#ff7fe1`.
- Copy: "Open your pack", "Tap to open", "Opening…", "Open the {pack name}", "Skip to the cards", "Pick one to keep".
- Conventional Commits; no AI attribution. Before each commit: `npx tsc -b && npm run lint && npx vitest run`.

## Review Focus

1. Pressing Enter/Space on the focused pack stage must open it once — not open (keydown) and then immediately skip (click). Test in Task 3.
2. Opening timers must never leak into the deal: skipping mid-tear, then the deal's own flip timings run from its mount. Test in Task 3.
3. Keyboard focus after the stage unmounts at `done` must land on the heading, not `<body>`. Test in Task 3.
4. Glare must never cover the photo window for any template, including shaped windows (bounding rect). Test in Task 5.
5. Reduced motion end to end: no stage, no glare element, CSS animations disabled. Tests in Tasks 3, 5, 6.

```ts
import { useCallback, useEffect, useState } from 'react'

export type OpenPhase = 'sealed' | 'shake' | 'tear' | 'burst' | 'done'

export const OPEN_TIMING = { auto: 3000, shake: 350, tear: 400, burst: 600 } as const

const NEXT: Record<Exclude<OpenPhase, 'done'>, OpenPhase> = { sealed: 'shake', shake: 'tear', tear: 'burst', burst: 'done' }

export const RARE_TINT = { holo: '#7fd6ff', poly: '#ff7fe1' } as const

/** Glow colour for the rarest edition in a pack (Polychrome beats Holographic), or null. */
export function rareTint(editionIds: string[]): string | null {
  if (editionIds.includes('poly')) return RARE_TINT.poly
  if (editionIds.includes('holo')) return RARE_TINT.holo
  return null
}

/** Pack-opening phases: sealed (until tap or 3 s) → shake → tear → burst → done. Reduced motion starts done. */
export function usePackOpen(reducedMotion: boolean) {
  const [phase, setPhase] = useState<OpenPhase>(reducedMotion ? 'done' : 'sealed')
  const open = useCallback(() => setPhase((p) => (p === 'sealed' ? 'shake' : p)), [])
  const skip = useCallback(() => setPhase('done'), [])

  useEffect(() => {
    if (phase === 'done') return
    const id = setTimeout(() => setPhase(NEXT[phase]), phase === 'sealed' ? OPEN_TIMING.auto : OPEN_TIMING[phase])
    return () => clearTimeout(id)
  }, [phase])

  // Any key opens while sealed and skips after; Enter/Space on a button is left to its click.
  useEffect(() => {
    if (phase === 'done') return
    const onKey = (e: KeyboardEvent) => {
      if (e.target instanceof HTMLButtonElement && (e.key === 'Enter' || e.key === ' ')) return
      if (phase === 'sealed') open()
      else skip()
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [phase, open, skip])

  return { phase, open, skip }
}
```

```ts
/** Art rows above this line are the crimped top strip that tears off when the pack opens. */
export const TEAR_ROW = 9

/** The pack art split into its top strip and the rest, both at `scale`; stacked they equal `packArt`. */
export function packArtParts(
  pack: PackDef,
  scale: number,
  targets: { top: HTMLCanvasElement; body: HTMLCanvasElement } = {
    top: document.createElement('canvas'),
    body: document.createElement('canvas'),
  },
) {
  const full = packArt(pack, scale)
  const split = TEAR_ROW * scale
  const parts = [
    [targets.top, 0, split],
    [targets.body, split, full.height - split],
  ] as const
  for (const [canvas, sy, h] of parts) {
    canvas.width = full.width
    canvas.height = h
    const ctx = canvas.getContext('2d')!
    ctx.imageSmoothingEnabled = false
    ctx.drawImage(full, 0, sy, full.width, h, 0, 0, full.width, h)
  }
  return targets
}
```

```tsx
import { useEffect, useRef, useState, type CSSProperties } from 'react'
import { getEdition } from '../editions/registry'
import { getPack, type PackDef } from '../packs/catalog'
import type { PackCard } from '../packs/deal'
import { packArtParts } from '../packs/packArt'
import type { PhotoSource } from '../photo/crop'
import { useBooth } from '../store/booth'
import { getTemplate } from '../templates/registry'
import { prefersReducedMotion } from '../ui/motion'
import { CardBackCanvas } from './CardBackCanvas'
import { StampCanvas } from './StampCanvas'
import { DEAL_TIMING, useDeal } from './useDeal'
import { rareTint, usePackOpen, type OpenPhase } from './usePackOpen'

const DEAL_SCALE = 2
const STAGE_SCALE = 4

function PackStage({ pack, phase, tint, onPress }: { pack: PackDef; phase: OpenPhase; tint: string | null; onPress: () => void }) {
  const top = useRef<HTMLCanvasElement>(null)
  const body = useRef<HTMLCanvasElement>(null)
  useEffect(() => {
    if (top.current && body.current) packArtParts(pack, STAGE_SCALE, { top: top.current, body: body.current })
  }, [pack])
  return (
    <>
      <button
        className="pack-stage"
        data-phase={phase}
        style={tint ? ({ '--glow': tint } as CSSProperties) : undefined}
        aria-label={phase === 'sealed' ? `Open the ${pack.name}` : 'Skip to the cards'}
        onClick={onPress}
      >
        <span className="pack-shaker">
          <canvas ref={top} className="pack-top" aria-hidden="true" />
          <canvas ref={body} className="pack-body" aria-hidden="true" />
          <span className="pack-flash" aria-hidden="true" />
        </span>
      </button>
      <p className="deal-hint">{phase === 'sealed' ? 'Tap to open' : 'Opening…'}</p>
    </>
  )
}

function DealCards({ pack, capture, reducedMotion, fromPack }: { pack: PackCard[]; capture: PhotoSource; reducedMotion: boolean; fromPack: boolean }) {
  const { pickPackCard } = useBooth.getState()
  const count = pack.length
  const { revealed, pickable, revealAll } = useDeal(count, reducedMotion)
  const [chosen, setChosen] = useState<number | null>(null)

  // Let the chosen card lift before switching screens.
  useEffect(() => {
    if (chosen === null) return
    const id = setTimeout(() => pickPackCard(chosen), reducedMotion ? 0 : DEAL_TIMING.choose)
    return () => clearTimeout(id)
  }, [chosen, reducedMotion, pickPackCard])

  const dealing = revealed < count
  return (
    <>
      <div className={`deal ${fromPack ? 'from-pack' : ''}`}>
        {/* existing pack.map(...) button markup, unchanged */}
      </div>
      <p className="deal-hint" aria-live="polite">{dealing ? 'Dealing…' : 'Tap a card to keep it'}</p>
    </>
  )
}

export function PackDeal() {
  const pack = useBooth((s) => s.pack)
  const capture = useBooth((s) => s.capture)
  const packId = useBooth((s) => s.packId)
  const { goToPicker } = useBooth.getState()
  const [reducedMotion] = useState(prefersReducedMotion)
  const { phase, open, skip } = usePackOpen(reducedMotion)
  const opening = phase !== 'done'
  const headingRef = useRef<HTMLHeadingElement>(null)

  // On arrival and again when the stage gives way to the cards (it may have had focus).
  useEffect(() => headingRef.current?.focus(), [opening])

  if (!pack || !capture) return null

  return (
    <section className="screen">
      <h2 ref={headingRef} tabIndex={-1}>{opening ? 'Open your pack' : 'Pick one to keep'}</h2>
      {opening ? (
        <PackStage
          pack={getPack(packId)}
          phase={phase}
          tint={rareTint(pack.map((c) => c.edition))}
          onPress={phase === 'sealed' ? open : skip}
        />
      ) : (
        <DealCards pack={pack} capture={capture} reducedMotion={reducedMotion} fromPack={!reducedMotion} />
      )}
      <button className="btn" onClick={goToPicker}>Back to shop</button>
    </section>
  )
}
```

`src/ui/TiltCard.tsx`:

```tsx
export function TiltCard({ children, delay = 0, motion = 'bob' }: { children: ReactNode; delay?: number; motion?: 'bob' | 'sway' }) {
  const ref = useRef<HTMLDivElement>(null)

  function set(rx: number, ry: number) {
    const el = ref.current
    if (!el) return
    el.style.setProperty('--rx', `${rx}deg`)
    el.style.setProperty('--ry', `${ry}deg`)
    // Unitless copies for shadows and glare (calc cannot turn degrees into pixels).
    el.style.setProperty('--tilt-x', String(ry))
    el.style.setProperty('--tilt-y', String(rx))
  }

  function onMove(e: PointerEvent<HTMLDivElement>) {
    const el = ref.current
    if (!el) return
    const { rx, ry } = tiltFor(e.clientX, e.clientY, el.getBoundingClientRect(), MAX_TILT)
    set(rx, ry)
  }

  return (
    <div ref={ref} className="tilt" onPointerMove={onMove} onPointerLeave={() => set(0, 0)}>
      <div className={`tilt-inner ${motion === 'sway' ? 'sway' : ''}`} style={{ animationDelay: `${delay}s` }}>
        {children}
      </div>
    </div>
  )
}
```

```ts
import type { Template } from '../templates/types'

export type GlareVariant = 'common' | 'foil' | 'holo' | 'poly'
const VARIANTS: readonly GlareVariant[] = ['foil', 'holo', 'poly']

export const glareVariant = (editionId: string): GlareVariant => VARIANTS.find((v) => v === editionId) ?? 'common'

const pct = (v: number, of: number) => `${Number(((v / of) * 100).toFixed(3))}%`

/** Card rectangle minus the photo window (its bounding rect for shaped windows), as a responsive clip-path. */
export function glareClip(t: Template): string {
  const { x, y, w, h } = t.photoRect
  const x1 = pct(x, t.size.w)
  const y1 = pct(y, t.size.h)
  const x2 = pct(x + w, t.size.w)
  const y2 = pct(y + h, t.size.h)
  return `polygon(evenodd, 0% 0%, 100% 0%, 100% 100%, 0% 100%, 0% 0%, ${x1} ${y1}, ${x2} ${y1}, ${x2} ${y2}, ${x1} ${y2}, ${x1} ${y1})`
}
```

```tsx
import { useRef, type CSSProperties, type ReactNode } from 'react'
import type { Edition } from '../editions/types'
import type { Template } from '../templates/types'
import { glareClip, glareVariant } from './glare'

const SPARKLES = 6

/** Display-only glare that follows the pointer and skips the photo window; shadow follows TiltCard's tilt. */
export function CardGlare({ template, edition, children }: { template: Template; edition: Edition; children: ReactNode }) {
  const ref = useRef<HTMLDivElement>(null)
  const variant = glareVariant(edition.id)
  return (
    <div
      ref={ref}
      className="card-glare"
      data-glare={variant}
      onPointerMove={(e) => {
        const el = ref.current
        if (!el) return
        const r = el.getBoundingClientRect()
        if (r.width <= 0 || r.height <= 0) return
        el.style.setProperty('--gx', `${Math.round(((e.clientX - r.left) / r.width) * 100)}%`)
        el.style.setProperty('--gy', `${Math.round(((e.clientY - r.top) / r.height) * 100)}%`)
        el.setAttribute('data-active', '')
      }}
      onPointerLeave={() => ref.current?.removeAttribute('data-active')}
    >
      {children}
      <div className="glare" aria-hidden="true" style={{ clipPath: glareClip(template) }}>
        {variant === 'poly' &&
          Array.from({ length: SPARKLES }, (_, i) => (
            <span key={i} className="sparkle-dot" style={{ '--s': i } as CSSProperties} />
          ))}
      </div>
    </div>
  )
}
```

# Sound Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give Retrato original synthesised background music and interaction sounds, with guest and operator controls.

**Architecture:** `src/audio/` holds a framework-free Web Audio engine (buses and a limiter), a table of effect recipes with `play()`, and a look-ahead music sequencer. React glue (`useAudio`, `useTapSounds`, `SoundButton`) is mounted in `App`. Animation hooks call `play()` at their steps. Settings live in a new `useSound` store (guest mode) and the existing operator store (volumes, button visibility).

**Tech Stack:** React 19, Zustand 5, TypeScript, Vitest 4 + jsdom (no Web Audio in jsdom, so tests use a fake AudioContext).

**Spec:** `docs/superpowers/specs/2026-10-15-sound-design.md`

## Global Constraints

- All audio is synthesised with the Web Audio API: no audio files, no new dependencies, and no Balatro audio, samples or transcriptions.
- No AudioContext available, or a constructor that throws, means a silent app with no errors and no speaker button.
- Nothing plays before the first gesture (`pointerdown`/`keydown`).
- Storage: the guest mode goes under `retrato:sound`; operator values go in the existing `retrato:operator`, with defaults music 40, effects 70 and showSoundButton true.
- At most 6 effects at once; the same effect within 40 ms is dropped.
- Music: 80 BPM, 8 bars, Dm9 Dm9 B♭maj7 B♭maj7 Gm9 Gm9 A7sus4 A7. Look-ahead 120 ms on a 25 ms tick. Bar times come from the bar index, and late bars are skipped.
- The commit style is Conventional Commits with no AI attribution. Run every step's command from `~/pixel-booth`.

## Review Focus

1. **Unlock race:** the first tap's `tick` may fire before `resume()` settles. It should simply be silent, never throw, and later taps should sound.
2. **Hidden tab or print dialog:** resuming must not burst a backlog of missed bars, and the scheduler must skip bars that already started (Task 3 test "late bars are skipped").
3. **Keyboard:** the **M** key must not fire while typing in editor fields or with the operator panel open (Task 7 tests).
4. **Saved data:** operator settings saved before this feature (no sound keys) must load with defaults and keep the other values (Task 4 test).
5. **Skipping animations:** skipping the pack opening or the print must stop sounds that are still ringing, not leave a whirr or burst playing (Tasks 9 and 10 tests).

```ts
// Test-only stand-in for the Web Audio API: records nodes, connections and scheduled values.
export interface ParamEvent { kind: 'set' | 'linear' | 'exp' | 'target' | 'cancel'; value: number; time: number }

export class FakeParam {
  value: number
  events: ParamEvent[] = []
  constructor(value = 0) {
    this.value = value
  }
  private rec(kind: ParamEvent['kind'], value: number, time: number) {
    this.events.push({ kind, value, time })
    return this
  }
  setValueAtTime(v: number, t: number) { this.value = v; return this.rec('set', v, t) }
  linearRampToValueAtTime(v: number, t: number) { this.value = v; return this.rec('linear', v, t) }
  exponentialRampToValueAtTime(v: number, t: number) { this.value = v; return this.rec('exp', v, t) }
  setTargetAtTime(v: number, t: number) { this.value = v; return this.rec('target', v, t) }
  cancelScheduledValues(t: number) { return this.rec('cancel', 0, t) }
}

export class FakeNode {
  kind: string
  outputs: (FakeNode | FakeParam)[] = []
  constructor(kind: string) {
    this.kind = kind
  }
  connect<T extends FakeNode | FakeParam>(n: T): T {
    this.outputs.push(n)
    return n
  }
  disconnect() {
    this.outputs = []
  }
}

export class FakeSource extends FakeNode {
  type = 'sine'
  frequency = new FakeParam(440)
  detune = new FakeParam(0)
  playbackRate = new FakeParam(1)
  buffer: unknown = null
  loop = false
  startedAt: number | null = null
  stoppedAt: number | null = null
  start(t = 0) { this.startedAt = t }
  stop(t = 0) { this.stoppedAt = t }
}

export class FakeAudioContext {
  currentTime = 0
  sampleRate = 8000
  state: 'suspended' | 'running' | 'closed' = 'suspended'
  destination = new FakeNode('destination')
  nodes: FakeNode[] = []
  private add<T extends FakeNode>(n: T): T {
    this.nodes.push(n)
    return n
  }
  createOscillator() { return this.add(new FakeSource('oscillator')) }
  createBufferSource() { return this.add(new FakeSource('buffer')) }
  createGain() { return this.add(Object.assign(new FakeNode('gain'), { gain: new FakeParam(1) })) }
  createBiquadFilter() {
    return this.add(Object.assign(new FakeNode('filter'), { type: 'lowpass', frequency: new FakeParam(350), Q: new FakeParam(1) }))
  }
  createDynamicsCompressor() {
    return this.add(Object.assign(new FakeNode('compressor'), {
      threshold: new FakeParam(-24), knee: new FakeParam(30), ratio: new FakeParam(12), attack: new FakeParam(0.003), release: new FakeParam(0.25),
    }))
  }
  createBuffer(_channels: number, length: number, sampleRate: number) {
    const data = new Float32Array(length)
    return { length, sampleRate, getChannelData: () => data }
  }
  resume() { this.state = 'running'; return Promise.resolve() }
  suspend() { this.state = 'suspended'; return Promise.resolve() }
  /** Oscillators and buffer sources created so far. */
  sources() { return this.nodes.filter((n): n is FakeSource => n instanceof FakeSource) }
}

export const asAudio = (c: FakeAudioContext) => c as unknown as AudioContext
```

```ts
// One shared Web Audio graph: music and effects buses → master → limiter → speakers.
export interface Levels {
  music: number
  sfx: number
}

export interface Engine {
  ctx: AudioContext
  master: GainNode
  /** Music volume: operator level × guest mode. */
  music: GainNode
  /** Per-screen music mix (full / quiet / duck). */
  musicMix: GainNode
  musicTone: BiquadFilterNode
  sfx: GainNode
  levels: Levels
  /** End times of effects still sounding (voice cap). */
  voices: number[]
  /** When each effect last started (same-sound guard). */
  lastPlayed: Map<string, number>
}

let engine: Engine | null | undefined

function build(ctx: AudioContext): Engine {
  const limiter = ctx.createDynamicsCompressor()
  limiter.threshold.value = -10
  limiter.ratio.value = 12
  limiter.attack.value = 0.003
  limiter.release.value = 0.25
  limiter.connect(ctx.destination)
  const master = ctx.createGain()
  master.connect(limiter)
  const sfx = ctx.createGain()
  sfx.gain.value = 0
  sfx.connect(master)
  const musicTone = ctx.createBiquadFilter()
  musicTone.type = 'lowpass'
  musicTone.frequency.value = 20000
  musicTone.connect(master)
  const musicMix = ctx.createGain()
  musicMix.connect(musicTone)
  const music = ctx.createGain()
  music.gain.value = 0
  music.connect(musicMix)
  return { ctx, master, music, musicMix, musicTone, sfx, levels: { music: 0, sfx: 0 }, voices: [], lastPlayed: new Map() }
}

function create(): Engine | null {
  const w = window as Window & { webkitAudioContext?: typeof AudioContext }
  const Ctor = w.AudioContext ?? w.webkitAudioContext
  if (!Ctor) return null
  try {
    return build(new Ctor())
  } catch {
    return null
  }
}

/** The shared engine, or null when this browser has no usable Web Audio (the app then stays silent). */
export function getEngine(): Engine | null {
  if (engine === undefined) engine = create()
  return engine
}

/** True once a gesture has let audio start, while the tab isn't paused. */
export const isRunning = () => getEngine()?.ctx.state === 'running'

/** Called on taps and key presses (browsers only start audio from one). Resolves to whether audio now runs. */
export async function unlock(): Promise<boolean> {
  const e = getEngine()
  if (!e) return false
  if (e.ctx.state !== 'running') {
    try {
      await e.ctx.resume()
    } catch {
      // Try again on the next gesture.
    }
  }
  return e.ctx.state === 'running'
}

export function setLevels(levels: Levels) {
  const e = getEngine()
  if (!e) return
  e.levels = levels
  const now = e.ctx.currentTime
  e.music.gain.setTargetAtTime(levels.music, now, 0.05)
  e.sfx.gain.setTargetAtTime(levels.sfx, now, 0.05)
}

export async function suspend() {
  try {
    await getEngine()?.ctx.suspend()
  } catch {
    // Stays as it was.
  }
}

export async function resume() {
  try {
    await getEngine()?.ctx.resume()
  } catch {
    // The next gesture retries.
  }
}

/** Test seam: a fake context, null for "no Web Audio", or undefined to recreate lazily. */
export function __setEngineForTest(ctx: AudioContext | null | undefined) {
  engine = ctx ? build(ctx) : ctx
}
```

```ts
import { getEngine } from './engine'

export type Sfx =
  | 'tick' | 'hover' | 'reroll' | 'shake' | 'tear' | 'burst' | 'flip' | 'rare1' | 'rare2' | 'rare3'
  | 'shutter' | 'flash' | 'whirr' | 'chime' | 'confirm' | 'bonk'

export interface Voice {
  stop(): void
}

export const MAX_VOICES = 6
/** Seconds: the same effect again sooner than this is dropped so rapid taps stay crisp. */
export const REPEAT_GUARD = 0.04

/** Edition → reveal arpeggio (rarer editions get more notes). */
export const EDITION_SFX: Record<string, Sfx> = { foil: 'rare1', holo: 'rare2', poly: 'rare3' }

type Rand = () => number
interface Ctx {
  ctx: BaseAudioContext
  out: AudioNode
  t: number
  rand: Rand
  nodes: AudioScheduledSourceNode[]
}

const hz = (midi: number) => 440 * 2 ** ((midi - 69) / 12)

const noiseBuffers = new WeakMap<BaseAudioContext, AudioBuffer>()
function noiseBuffer(ctx: BaseAudioContext): AudioBuffer {
  let b = noiseBuffers.get(ctx)
  if (!b) {
    b = ctx.createBuffer(1, ctx.sampleRate, ctx.sampleRate)
    const d = b.getChannelData(0)
    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1
    noiseBuffers.set(ctx, b)
  }
  return b
}

/** Gain that rises to `peak` and decays to silence by `end`. */
function envelope(c: Ctx, peak: number, start: number, end: number, attack = 0.002): GainNode {
  const g = c.ctx.createGain()
  g.gain.setValueAtTime(0.0001, start)
  g.gain.linearRampToValueAtTime(peak, start + attack)
  g.gain.exponentialRampToValueAtTime(0.0001, end)
  g.connect(c.out)
  return g
}

interface ToneOpts { type: OscillatorType; freq: number; to?: number; at?: number; dur: number; gain: number; attack?: number }
function tone(c: Ctx, o: ToneOpts) {
  const start = c.t + (o.at ?? 0)
  const end = start + o.dur
  const osc = c.ctx.createOscillator()
  osc.type = o.type
  osc.frequency.setValueAtTime(o.freq, start)
  if (o.to) osc.frequency.exponentialRampToValueAtTime(o.to, end)
  osc.connect(envelope(c, o.gain, start, end, o.attack))
  osc.start(start)
  osc.stop(end + 0.02)
  c.nodes.push(osc)
}

interface NoiseOpts { at?: number; dur: number; gain: number; filter: BiquadFilterType; freq: number; to?: number; q?: number; attack?: number }
function noise(c: Ctx, o: NoiseOpts) {
  const start = c.t + (o.at ?? 0)
  const end = start + o.dur
  const src = c.ctx.createBufferSource()
  src.buffer = noiseBuffer(c.ctx)
  const f = c.ctx.createBiquadFilter()
  f.type = o.filter
  f.frequency.setValueAtTime(o.freq, start)
  if (o.to) f.frequency.exponentialRampToValueAtTime(o.to, end)
  if (o.q) f.Q.value = o.q
  src.connect(f)
  f.connect(envelope(c, o.gain, start, end, o.attack))
  src.start(start)
  src.stop(end + 0.02)
  c.nodes.push(src)
}

/** The same context, but sounding through a lowpass filter. */
function lowpassed(c: Ctx, freq: number): Ctx {
  const f = c.ctx.createBiquadFilter()
  f.type = 'lowpass'
  f.frequency.value = freq
  f.connect(c.out)
  return { ...c, out: f }
}

function arpeggio(c: Ctx, count: number) {
  const notes = [72, 76, 79, 84, 88, 91].slice(0, count) // C5 E5 G5 C6 E6 G6
  notes.forEach((m, i) => tone(c, { type: 'square', freq: hz(m), at: i * 0.07, dur: 0.12, gain: 0.1 }))
  return (count - 1) * 0.07 + 0.12
}

// Each recipe schedules its sources from c.t and returns its length in seconds.
const RECIPES: Record<Sfx, (c: Ctx) => number> = {
  tick: (c) => {
    tone(c, { type: 'square', freq: 1200 * (0.92 + c.rand() * 0.16), dur: 0.04, gain: 0.18 })
    return 0.045
  },
  hover: (c) => {
    tone(c, { type: 'triangle', freq: 2400, dur: 0.025, gain: 0.06 })
    return 0.03
  },
  reroll: (c) => {
    ;[84, 86, 88, 91, 93, 96].forEach((m, i) => tone(c, { type: 'square', freq: hz(m), at: i * 0.03, dur: 0.05, gain: 0.12 }))
    return 0.2
  },
  shake: (c) => {
    for (let i = 0; i < 4; i++) noise(c, { at: i * 0.06, dur: 0.05, gain: 0.25, filter: 'bandpass', freq: 800, q: 4 })
    return 0.26
  },
  tear: (c) => {
    noise(c, { dur: 0.22, gain: 0.3, filter: 'highpass', freq: 600, to: 4000, attack: 0.01 })
    return 0.22
  },
  burst: (c) => {
    ;[72, 76, 79, 83].forEach((m, i) => tone(c, { type: 'triangle', freq: hz(m), at: i * 0.015, dur: 0.6, gain: 0.1 }))
    for (let i = 0; i < 3; i++)
      tone(c, { type: 'square', freq: hz(96 + Math.floor(c.rand() * 5) * 2), at: 0.25 + i * 0.08, dur: 0.08, gain: 0.05 })
    return 0.7
  },
  flip: (c) => {
    noise(c, { dur: 0.2, gain: 0.2, filter: 'lowpass', freq: 3000, to: 800, attack: 0.02 })
    tone(c, { type: 'triangle', freq: 140, to: 90, at: 0.12, dur: 0.12, gain: 0.3 })
    return 0.25
  },
  rare1: (c) => arpeggio(c, 3),
  rare2: (c) => arpeggio(c, 4),
  rare3: (c) => arpeggio(c, 6),
  shutter: (c) => {
    noise(c, { dur: 0.03, gain: 0.3, filter: 'highpass', freq: 2000 })
    noise(c, { at: 0.07, dur: 0.04, gain: 0.25, filter: 'highpass', freq: 2000 })
    tone(c, { type: 'sine', freq: 90, dur: 0.1, gain: 0.4 })
    return 0.15
  },
  flash: (c) => {
    tone(c, { type: 'sine', freq: 3000, to: 300, dur: 0.3, gain: 0.15 })
    noise(c, { dur: 0.25, gain: 0.08, filter: 'highpass', freq: 5000 })
    return 0.3
  },
  whirr: (c) => {
    // Motor: a buzzy saw through a lowpass, with a fast tremolo.
    const end = c.t + 0.9
    const lp = lowpassed({ ...c, out: envelope(c, 0.12, c.t, end, 0.1) }, 500)
    const trem = c.ctx.createGain()
    trem.gain.value = 0.5
    trem.connect(lp.out)
    const osc = c.ctx.createOscillator()
    osc.type = 'sawtooth'
    osc.frequency.setValueAtTime(70, c.t)
    osc.connect(trem)
    const lfo = c.ctx.createOscillator()
    lfo.frequency.setValueAtTime(20, c.t)
    const depth = c.ctx.createGain()
    depth.gain.value = 0.5
    lfo.connect(depth)
    depth.connect(trem.gain)
    for (const s of [osc, lfo]) {
      s.start(c.t)
      s.stop(end + 0.02)
      c.nodes.push(s)
    }
    return 0.9
  },
  chime: (c) => {
    tone(c, { type: 'triangle', freq: hz(79), dur: 0.5, gain: 0.15 })
    tone(c, { type: 'triangle', freq: hz(84), at: 0.12, dur: 0.48, gain: 0.15 })
    return 0.6
  },
  confirm: (c) => {
    tone(c, { type: 'square', freq: hz(88), dur: 0.07, gain: 0.1 })
    tone(c, { type: 'square', freq: hz(95), at: 0.06, dur: 0.09, gain: 0.1 })
    return 0.15
  },
  bonk: (c) => {
    const lp = lowpassed(c, 1200)
    tone(lp, { type: 'square', freq: hz(50), dur: 0.1, gain: 0.15 })
    tone(lp, { type: 'square', freq: hz(45), at: 0.09, dur: 0.15, gain: 0.15 })
    return 0.25
  },
}

export const SFX_NAMES = Object.keys(RECIPES) as Sfx[]

/** Schedules one effect on any context, live or offline (the listening preview uses this too). */
export function renderSfx(ctx: BaseAudioContext, out: AudioNode, name: Sfx, t: number, rand: Rand = Math.random): Voice & { length: number } {
  const c: Ctx = { ctx, out, t, rand, nodes: [] }
  const length = RECIPES[name](c)
  let stopped = false
  return {
    length,
    stop() {
      if (stopped) return
      stopped = true
      for (const n of c.nodes) {
        try {
          n.stop()
        } catch {
          // Already ended.
        }
      }
    },
  }
}

/** Plays an effect now; null (and silence) when audio is locked, muted, too busy or just played. */
export function play(name: Sfx, opts: { rand?: Rand } = {}): Voice | null {
  const e = getEngine()
  if (!e || e.ctx.state !== 'running' || e.levels.sfx <= 0) return null
  const now = e.ctx.currentTime
  const last = e.lastPlayed.get(name)
  if (last !== undefined && now - last < REPEAT_GUARD) return null
  e.voices = e.voices.filter((end) => end > now)
  if (e.voices.length >= MAX_VOICES) return null
  const voice = renderSfx(e.ctx, e.sfx, name, now, opts.rand)
  e.lastPlayed.set(name, now)
  e.voices.push(now + voice.length)
  return voice
}
```

```ts
import { getEngine, type Engine } from './engine'

// An original slow, dreamy loop: detuned pads with a slow pitch wobble, a soft bass and a sparse melody.
export const BPM = 80
export const BEAT = 60 / BPM
export const BAR = BEAT * 4
/** Seconds ahead the scheduler looks on each tick. */
export const LOOKAHEAD = 0.12
export const TICK_MS = 25

interface Chord {
  name: string
  /** MIDI note of the bass. */
  bass: number
  /** MIDI notes of the pad voicing. */
  tones: readonly number[]
}
const DM9: Chord = { name: 'Dm9', bass: 38, tones: [50, 53, 57, 60, 64] }
const BBMAJ7: Chord = { name: 'B♭maj7', bass: 34, tones: [50, 53, 57, 58] }
const GM9: Chord = { name: 'Gm9', bass: 43, tones: [46, 50, 53, 55, 57] }
const A7SUS4: Chord = { name: 'A7sus4', bass: 45, tones: [52, 55, 57, 62] }
const A7: Chord = { name: 'A7', bass: 45, tones: [52, 55, 57, 61] }
export const PROGRESSION: readonly Chord[] = [DM9, DM9, BBMAJ7, BBMAJ7, GM9, GM9, A7SUS4, A7]

/** D minor pentatonic, octave 5. */
export const MELODY_NOTES = [74, 77, 79, 81, 84] as const
export const MELODY_CHANCE = 0.35

export const hz = (midi: number) => 440 * 2 ** ((midi - 69) / 12)
/** Computed from the bar index, never accumulated, so the loop can't drift. */
export const barStart = (loopStart: number, bar: number) => loopStart + bar * BAR

function note(ctx: BaseAudioContext, out: AudioNode, nodes: AudioScheduledSourceNode[], type: OscillatorType, freq: number, t: number, dur: number, gain: number) {
  const g = ctx.createGain()
  g.gain.setValueAtTime(0.0001, t)
  g.gain.linearRampToValueAtTime(gain, t + 0.01)
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  g.connect(out)
  const o = ctx.createOscillator()
  o.type = type
  o.frequency.setValueAtTime(freq, t)
  o.connect(g)
  o.start(t)
  o.stop(t + dur + 0.02)
  nodes.push(o)
}

/** Schedules bar `bar` (the progression repeats every 8) from time t. Works on live and offline contexts. */
export function scheduleBar(ctx: BaseAudioContext, out: AudioNode, wobble: AudioNode | null, bar: number, t: number, rand: () => number) {
  const chord = PROGRESSION[bar % PROGRESSION.length]
  const nodes: AudioScheduledSourceNode[] = []
  // Pads: two detuned saws per chord tone, slow swell and release; the wobble LFO bends them together.
  const padEnd = t + BAR + 1.2
  const pad = ctx.createGain()
  pad.gain.setValueAtTime(0.0001, t)
  pad.gain.linearRampToValueAtTime(0.035, t + 0.8)
  pad.gain.setValueAtTime(0.035, t + BAR)
  pad.gain.linearRampToValueAtTime(0.0001, padEnd)
  const lp = ctx.createBiquadFilter()
  lp.type = 'lowpass'
  lp.frequency.value = 1400
  pad.connect(lp)
  lp.connect(out)
  for (const m of chord.tones) {
    for (const cents of [-7, 7]) {
      const o = ctx.createOscillator()
      o.type = 'sawtooth'
      o.frequency.setValueAtTime(hz(m), t)
      o.detune.setValueAtTime(cents, t)
      wobble?.connect(o.detune)
      o.connect(pad)
      o.start(t)
      o.stop(padEnd + 0.05)
      nodes.push(o)
    }
  }
  for (const beat of [0, 2]) note(ctx, out, nodes, 'triangle', hz(chord.bass), t + beat * BEAT, 1.2, 0.22)
  for (let beat = 0; beat < 4; beat++) {
    if (rand() >= MELODY_CHANCE) continue
    const m = MELODY_NOTES[Math.floor(rand() * MELODY_NOTES.length)]
    note(ctx, out, nodes, 'square', hz(m), t + beat * BEAT, 0.35, 0.05)
  }
  return nodes
}

interface Session {
  ctx: AudioContext
  out: GainNode
  lfo: OscillatorNode
  wobble: GainNode
  loopStart: number
  nextBar: number
  timer: ReturnType<typeof setInterval>
  /** Sources of the last few bars, so stop can silence them. */
  bars: AudioScheduledSourceNode[][]
  rand: () => number
}
let session: Session | null = null

/** Schedules every bar starting inside the look-ahead window; a bar whose start already passed is skipped. */
function pump(s: Session) {
  const now = s.ctx.currentTime
  while (barStart(s.loopStart, s.nextBar) < now + LOOKAHEAD) {
    const t = barStart(s.loopStart, s.nextBar)
    if (t >= now) {
      s.bars.push(scheduleBar(s.ctx, s.out, s.wobble, s.nextBar, t, s.rand))
      if (s.bars.length > 3) s.bars.shift()
    }
    s.nextBar++
  }
}

/** Fades the loop in over 2 s. Does nothing without running audio or when already playing. */
export function startMusic(rand: () => number = Math.random) {
  const e = getEngine()
  if (!e || session || e.ctx.state !== 'running') return
  const ctx = e.ctx
  const now = ctx.currentTime
  const out = ctx.createGain()
  out.gain.setValueAtTime(0.0001, now)
  out.gain.linearRampToValueAtTime(1, now + 2)
  out.connect(e.music)
  const lfo = ctx.createOscillator()
  lfo.frequency.value = 0.25
  const wobble = ctx.createGain()
  wobble.gain.value = 12 // ±12 cents
  lfo.connect(wobble)
  lfo.start(now)
  const s: Session = { ctx, out, lfo, wobble, loopStart: now + 0.05, nextBar: 0, bars: [], rand, timer: 0 as unknown as ReturnType<typeof setInterval> }
  session = s
  pump(s)
  s.timer = setInterval(() => pump(s), TICK_MS)
}

/** Fades out over 0.5 s and ends the loop. */
export function stopMusic() {
  const s = session
  if (!s) return
  session = null
  clearInterval(s.timer)
  const now = s.ctx.currentTime
  s.out.gain.cancelScheduledValues(now)
  s.out.gain.setValueAtTime(s.out.gain.value, now)
  s.out.gain.linearRampToValueAtTime(0.0001, now + 0.5)
  for (const n of [s.lfo, ...s.bars.flat()]) {
    try {
      n.stop(now + 0.55)
    } catch {
      // Already ended.
    }
  }
}

export const isMusicPlaying = () => session !== null

export type Mix = 'full' | 'quiet' | 'duck'
export const MIX = { full: { level: 1, tone: 20000 }, quiet: { level: 0.55, tone: 900 }, duck: { level: 0.3, tone: 0 } } as const
export const DUCK_MS = 1200
let base: Exclude<Mix, 'duck'> = 'full'
let duckTimer: ReturnType<typeof setTimeout> | undefined

function apply(e: Engine, mix: Mix) {
  const now = e.ctx.currentTime
  e.musicMix.gain.setTargetAtTime(MIX[mix].level, now, 0.15)
  e.musicTone.frequency.setTargetAtTime(mix === 'duck' ? MIX[base].tone : MIX[mix].tone, now, 0.15)
}

/** Screen mix (full in the shop, quiet elsewhere); duck dips under a loud effect, then returns. */
export function setMix(mix: Mix) {
  const e = getEngine()
  if (!e) return
  if (mix !== 'duck') base = mix
  clearTimeout(duckTimer)
  apply(e, mix)
  if (mix === 'duck') duckTimer = setTimeout(() => apply(e, base), DUCK_MS)
}
```

```ts
import { useEffect, useState } from 'react'
import { useBooth } from '../store/booth'
import { useOperator } from '../store/operator'
import { effectiveLevels, useSound } from '../store/sound'
import { useTray } from '../store/tray'
import { getEngine, isRunning, resume, setLevels, suspend, unlock } from './engine'
import { setMix, startMusic, stopMusic } from './music'

/** App-wide audio: unlock on a gesture, follow the volumes, play music, mix per screen, pause when hidden. */
export function useAudio() {
  const mode = useSound((s) => s.mode)
  const musicVolume = useOperator((s) => s.musicVolume)
  const sfxVolume = useOperator((s) => s.sfxVolume)
  const screen = useBooth((s) => s.screen)
  const trayOpen = useTray((s) => s.open)
  const [running, setRunning] = useState(false)
  const { music, sfx } = effectiveLevels(mode, musicVolume, sfxVolume)
  const musicOn = music > 0

  // Browsers only start audio from a gesture, so try on every tap or key until it runs.
  useEffect(() => {
    if (!getEngine()) return
    const onGesture = () => {
      if (isRunning() || document.visibilityState === 'hidden') return
      void unlock().then(setRunning)
    }
    document.addEventListener('pointerdown', onGesture, true)
    document.addEventListener('keydown', onGesture, true)
    return () => {
      document.removeEventListener('pointerdown', onGesture, true)
      document.removeEventListener('keydown', onGesture, true)
    }
  }, [])

  useEffect(() => setLevels({ music, sfx }), [music, sfx])

  useEffect(() => {
    if (running && musicOn) startMusic()
    else stopMusic()
  }, [running, musicOn])

  useEffect(() => setMix(trayOpen || screen !== 'picker' ? 'quiet' : 'full'), [screen, trayOpen])

  // A hidden tab or the print dialog pauses everything; it picks up where it was afterwards.
  useEffect(() => {
    if (!getEngine()) return
    let paused = false
    const pause = () => {
      if (!isRunning()) return
      paused = true
      void suspend()
    }
    const pickUp = () => {
      if (!paused) return
      paused = false
      void resume()
    }
    const onVisibility = () => (document.visibilityState === 'hidden' ? pause() : pickUp())
    document.addEventListener('visibilitychange', onVisibility)
    window.addEventListener('beforeprint', pause)
    window.addEventListener('afterprint', pickUp)
    return () => {
      document.removeEventListener('visibilitychange', onVisibility)
      window.removeEventListener('beforeprint', pause)
      window.removeEventListener('afterprint', pickUp)
    }
  }, [])

  useEffect(() => () => stopMusic(), [])
}
```

```ts
import { useEffect } from 'react'
import { play } from './sfx'

const TAPPABLE = 'button, select, summary, [role="button"], .tile-btn'
const HOVERABLE = '.tile-btn, .pack-btn'

/** A tap on something clickable that doesn't play its own sound (marked data-sfx). */
export function isTappable(target: EventTarget | null): boolean {
  if (!(target instanceof Element)) return false
  const hit = target.closest(TAPPABLE)
  if (!hit || hit.closest('[data-sfx]')) return false
  return !(hit as HTMLButtonElement).disabled && hit.getAttribute('aria-disabled') !== 'true'
}

/** One page-wide listener so every button ticks without each one wiring a sound. */
export function useTapSounds() {
  useEffect(() => {
    const onDown = (e: PointerEvent) => {
      if (isTappable(e.target)) play('tick')
    }
    const onOver = (e: PointerEvent) => {
      if (!window.matchMedia?.('(pointer: fine)').matches) return
      const tile = e.target instanceof Element ? e.target.closest(HOVERABLE) : null
      if (tile && !(e.relatedTarget instanceof Node && tile.contains(e.relatedTarget))) play('hover')
    }
    document.addEventListener('pointerdown', onDown, true)
    document.addEventListener('pointerover', onOver, true)
    return () => {
      document.removeEventListener('pointerdown', onDown, true)
      document.removeEventListener('pointerover', onOver, true)
    }
  }, [])
}
```

```tsx
// Pixel speaker on an 11×9 grid: body plus waves (all), a note (effects only), or an X (off).
const BODY = 'M0 3h2v3H0zM2 2h1v5H2zM3 1h1v7H3zM4 0h1v9H4z'
const MARK: Record<SoundMode, string> = {
  all: 'M6 3h1v3H6zM8 1h1v1H8zM9 2h1v5H9zM8 7h1v1H8z',
  sfx: 'M7 1h3v1H7zM9 2h1v4H9zM7 5h2v2H7z',

}

function typing(target: EventTarget | null) {
  return target instanceof HTMLElement && (target.matches('input, textarea, select') || target.isContentEditable)
}

/** Guest sound toggle; M cycles it too unless keysOff (operator panel open) or the guest is typing. */
export function SoundButton({ keysOff }: { keysOff: boolean }) {
  const mode = useSound((s) => s.mode)
  const show = useOperator((s) => s.showSoundButton)
  const available = show && getEngine() !== null

    if (!available || keysOff) return
    const onKey = (e: KeyboardEvent) => {
      if (e.key.toLowerCase() !== 'm' || e.ctrlKey || e.metaKey || e.altKey || typing(e.target)) return
      useSound.getState().cycle()
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [available, keysOff])

  if (!available) return null
  return (

        <path d={BODY} />
        <path d={MARK[mode]} />

}
```
